def make_unique_names(raw_col_names: List[str]) -> List[str]:
    seen = {}
    unique_col_names = []
    for name in raw_col_names:
        s = str(name).strip()
        if s not in seen:
            seen[s] = 1
            unique_col_names.append(s)
        else:
            seen[s] += 1
            unique_col_names.append(f"{s}_{seen[s]}")
    return unique_col_names


def load_sheet_with_meta(file_path: str, sheet_idx: int) -> Tuple[pd.DataFrame, Dict]:
    df_raw = pd.read_excel(file_path, sheet_name=sheet_idx, header=None)

    raw_codes = [str(x).strip() for x in df_raw.iloc[0, 1:].tolist()]
    raw_names = [str(x).strip() for x in df_raw.iloc[1, 1:].tolist()]
    unique_names = make_unique_names(raw_names)

    data = df_raw.iloc[2:, :len(unique_names) + 1].copy()
    data.columns = ["Date"] + unique_names
    data["Date"] = pd.to_datetime(data["Date"], errors="coerce")
    data = data.dropna(subset=["Date"])

    df = data.set_index("Date").sort_index()
    df = df[~df.index.duplicated(keep="first")]

    for c in df.columns:
        df[c] = pd.to_numeric(df[c], errors="coerce")

    df = df.ffill().fillna(0.0)

    meta = {
        "sheet_idx": sheet_idx,
        "codes": raw_codes[:len(unique_names)],
        "raw_names": raw_names[:len(unique_names)],
        "unique_names": unique_names,
        "sheet_name": f"Sheet{sheet_idx + 1}",
    }
    return df, meta


def load_and_clean_sheet(file_path: str, sheet_idx: int) -> pd.DataFrame:
    df, _ = load_sheet_with_meta(file_path, sheet_idx)
    return df

THS_FETCH_SUMMARY: List[Dict] = []


def reset_ths_fetch_summary() -> None:
    global THS_FETCH_SUMMARY
    THS_FETCH_SUMMARY = []


def append_ths_fetch_summary(
    sheet_name: str,
    index_name: str,
    code: str,
    fetch_method: str,
    fetch_start: str,
    fetch_end: str,
    ser: pd.Series,
) -> None:
    global THS_FETCH_SUMMARY
    if ser is None or ser.empty:
        return
    latest_trade_date = pd.to_datetime(ser.index.max(), errors="coerce")
    latest_close = pd.to_numeric(ser.loc[latest_trade_date], errors="coerce") if pd.notna(latest_trade_date) else np.nan
    THS_FETCH_SUMMARY.append({
        "sheet_name": sheet_name,
        "指数名称": index_name,
        "指数代码": code,
        "取数方式": fetch_method,
        "取数起始": fetch_start,
        "取数截止": fetch_end,
        "最新交易日": latest_trade_date.strftime("%Y-%m-%d") if pd.notna(latest_trade_date) else "",
        "最新点位": float(latest_close) if pd.notna(latest_close) else np.nan,
        "取到记录数": int(len(ser)),
    })


def get_ths_fetch_summary_df() -> pd.DataFrame:
    cols = ["sheet_name", "指数名称", "指数代码", "取数方式", "取数起始", "取数截止", "最新交易日", "最新点位", "取到记录数"]
    if not THS_FETCH_SUMMARY:
        return pd.DataFrame(columns=cols)
    df = pd.DataFrame(THS_FETCH_SUMMARY)

    df = df[cols].copy()
    df = df.sort_values(["sheet_name", "指数代码", "指数名称"]).reset_index(drop=True)
    return df

def _excel_safe_value(x):
    if pd.isna(x):
        return None
    if isinstance(x, pd.Timestamp):
        return x.strftime("%Y-%m-%d")
    if hasattr(x, "item") and callable(getattr(x, "item")):
        try:
            return x.item()
        except Exception:
            return x
    return x


def normalize_date_str(x) -> str:
    if x is None or str(x).strip() == "":
        return ""
    return pd.Timestamp(x).strftime("%Y-%m-%d")


def get_target_end_date() -> pd.Timestamp:
    return pd.Timestamp(THS_END_DATE).normalize() if THS_END_DATE else pd.Timestamp.today().normalize()


def get_free_window_start(end_date: pd.Timestamp) -> pd.Timestamp:
    return (end_date - pd.DateOffset(years=THS_FREE_YEARS_LIMIT)).normalize()


def infer_fetch_window(local_df: pd.DataFrame, mode: str) -> Tuple[str, str]:
    end_date = get_target_end_date()
    free_start = get_free_window_start(end_date)

    mode = mode.upper()
    if mode == "THS_ONLY":
        start_date = pd.Timestamp(THS_START_DATE).normalize() if THS_START_DATE else free_start
        start_date = max(start_date, free_start)
    elif mode == "LOCAL_PLUS_THS":
        if local_df is not None and not local_df.empty:
            start_date = (local_df.index.max() - pd.Timedelta(days=THS_OVERLAP_DAYS)).normalize()
        else:
            start_date = free_start
        if THS_START_DATE:
            start_date = max(start_date, pd.Timestamp(THS_START_DATE).normalize())
        start_date = max(start_date, free_start)
    else:
        return "", ""

    if start_date > end_date:
        return "", ""

    return start_date.strftime("%Y-%m-%d"), end_date.strftime("%Y-%m-%d")

def import_ifind_sdk():
    try:
        return _try_import_ifind_sdk()
    except ImportError:
        pass

    # 允许手动把同花顺 SDK 目录塞进 sys.path，避免环境未修复时直接报错
    candidate_paths = []
    if IFIND_IMPORT_PATH:
        candidate_paths.append(IFIND_IMPORT_PATH)

    for raw_path in candidate_paths:
        p = os.path.abspath(os.path.expanduser(str(raw_path)))
        if not os.path.isdir(p):
            continue
        if p not in sys.path:
            sys.path.insert(0, p)
        try:
            return _try_import_ifind_sdk()
        except ImportError:
            continue

    raise ImportError(
        "未检测到 iFinDPy。当前不是登录失败，而是 Python 环境没有成功导入同花顺 SDK。\n"
        "请优先在同花顺 SuperCommand / 超级命令里修复当前 Python 解释器环境；\n"
        "若仍不行，可在参数区填写 IFIND_IMPORT_PATH（指向含 iFinDPy 的目录）；\n"
        "若你暂时只想跑本地模板，把 DATA_SOURCE_MODE 改成 LOCAL_ONLY。"
    )


def ths_login(sdk: Dict) -> None:
    if not THS_USERNAME or not THS_PASSWORD:
        raise ValueError("DATA_SOURCE_MODE 需要同花顺账号密码。请填写 THS_USERNAME / THS_PASSWORD，或改用环境变量。")
    ret = sdk["THS_iFinDLogin"](THS_USERNAME, THS_PASSWORD)
    if ret not in (0, -201):
        raise RuntimeError(f"同花顺登录失败，返回码={ret}")


def ths_logout(sdk: Dict) -> None:
    try:
        sdk["THS_iFinDLogout"]()
    except Exception:
        pass


def coerce_ths_result_to_df(raw, trans2df):
    if hasattr(raw, "errorcode") and getattr(raw, "errorcode") not in (0, None):
        errmsg = getattr(raw, "errmsg", "未知错误")
        raise RuntimeError(f"同花顺接口报错: {errmsg}")

    if hasattr(raw, "data") and isinstance(raw.data, pd.DataFrame):
        return raw.data.copy()

    if trans2df is not None:
        try:
            df = trans2df(raw)
            if isinstance(df, pd.DataFrame):
                return df.copy()
        except Exception:
            pass

    data = getattr(raw, "data", None)
    if isinstance(data, pd.DataFrame):
        return data.copy()
    if isinstance(data, dict):
        return pd.DataFrame(data)
    if isinstance(raw, dict):
        if "tables" in raw and raw["tables"]:
            table = raw["tables"][0].get("table", raw["tables"][0])
            return pd.DataFrame(table)
        return pd.DataFrame(raw)

    raw_text = data if isinstance(data, str) else raw if isinstance(raw, str) else None
    if raw_text:
        try:
            obj = json.loads(raw_text)
            if isinstance(obj, dict) and "tables" in obj and obj["tables"]:
                table = obj["tables"][0].get("table", obj["tables"][0])
                return pd.DataFrame(table)
            return pd.DataFrame(obj)
        except Exception as e:
            raise TypeError(f"无法解析同花顺返回结果，原始内容片段: {str(raw_text)[:200]}") from e

    raise TypeError(f"无法识别的同花顺返回类型: {type(raw)}")


def extract_single_series_from_df(df: pd.DataFrame, preferred_value_col: str, code: str) -> pd.Series:
    if df is None or df.empty:
        return pd.Series(dtype=float, name=code)

    work = df.copy()

    date_col = None
    for c in ["time", "Date", "date", "datetime"]:
        if c in work.columns:
            date_col = c
            break

    if date_col is None:
        if isinstance(work.index, pd.DatetimeIndex):
            idx = work.index
        else:
            idx = pd.to_datetime(work.index, errors="coerce")
    else:
        idx = pd.to_datetime(work[date_col], errors="coerce")

    exclude_cols = {date_col, "thscode", "THSCODE", "security_code", "symbol", "code", None}

    if preferred_value_col in work.columns:
        value_col = preferred_value_col
    else:
        close_like = [c for c in work.columns if "close" in str(c).lower() and c not in exclude_cols]
        if close_like:
            value_col = close_like[0]
        else:
            numeric_cols = []
            for c in work.columns:
                if c in exclude_cols:
                    continue
                s = pd.to_numeric(work[c], errors="coerce")
                if s.notna().any():
                    numeric_cols.append(c)
            if not numeric_cols:
                raise ValueError(f"{code} 返回数据中未找到可用数值列，列名={list(work.columns)}")
            value_col = numeric_cols[-1]

    ser = pd.Series(pd.to_numeric(work[value_col], errors="coerce").values, index=idx, name=code)
    ser = ser[ser.index.notna()]
    ser = ser[~ser.index.duplicated(keep="last")].sort_index()
    return ser


def fetch_ths_close_prices(meta: Dict, start_date: str, end_date: str) -> Tuple[pd.DataFrame, Dict[str, str]]:
    if not start_date or not end_date:
        return pd.DataFrame(columns=meta["unique_names"]), {}

    sdk = import_ifind_sdk()
    ths_login(sdk)

    out = {}
    errors = {}

    try:
        for code, name in zip(meta["codes"], meta["unique_names"]):
            try:
                raw = sdk["THS_DateSerial"](
                    code,
                    THS_INDEX_CLOSE_FIELD,
                    "",
                    THS_DATE_SERIAL_PARAM,
                    start_date,
                    end_date,
                    True,
                )
                df_raw = coerce_ths_result_to_df(raw, sdk.get("THS_Trans2DataFrame"))
                ser = extract_single_series_from_df(df_raw, THS_INDEX_CLOSE_FIELD, code)
                ser.name = name

                if ser.empty:
                    raise ValueError("日期序列函数返回为空")

                out[name] = ser
                append_ths_fetch_summary(meta.get("sheet_name", ""), name, code, "THS_DateSerial", start_date, end_date, ser)
                print(f"✅ 同花顺更新成功: {name} ({code}) | {start_date} ~ {end_date} | {len(ser)} 条")
            except Exception as e1:
                try:
                    raw = sdk["THS_HistoryQuotes"](
                        code,
                        THS_HQ_INDICATOR,
                        THS_HQ_PARAM,
                        start_date,
                        end_date,
                    )
                    df_raw = coerce_ths_result_to_df(raw, sdk.get("THS_Trans2DataFrame"))
                    ser = extract_single_series_from_df(df_raw, "close", code)
                    ser.name = name

                    if ser.empty:
                        raise ValueError("历史行情函数返回为空")

                    out[name] = ser
                    append_ths_fetch_summary(meta.get("sheet_name", ""), name, code, "THS_HistoryQuotes", start_date, end_date, ser)
                    print(f"✅ 同花顺回退成功: {name} ({code}) | {start_date} ~ {end_date} | {len(ser)} 条")
                except Exception as e2:
                    errors[name] = f"DateSerial失败: {e1}; HistoryQuotes失败: {e2}"
                    print(f"⚠️ {name} ({code}) 更新失败，保留本地数据。原因：{errors[name]}")
    finally:
        ths_logout(sdk)

    if not out:
        return pd.DataFrame(columns=meta["unique_names"]), errors

    api_df = pd.concat(out.values(), axis=1)
    api_df = api_df.reindex(columns=meta["unique_names"])
    api_df.index = pd.to_datetime(api_df.index)
    api_df = api_df.sort_index()
    api_df = api_df.ffill()
    return api_df, errors


def merge_local_and_api(local_df: pd.DataFrame, api_df: pd.DataFrame, meta: Dict, mode: str) -> pd.DataFrame:
    mode = mode.upper()

    if mode == "THS_ONLY":
        merged = api_df.copy()
    elif mode == "LOCAL_PLUS_THS":
        if local_df is None or local_df.empty:
            merged = api_df.copy()
        elif api_df is None or api_df.empty:
            merged = local_df.copy()
        else:
            merged = pd.concat([local_df, api_df], axis=0).sort_index()
            merged = merged[~merged.index.duplicated(keep="last")]
    else:
        merged = local_df.copy()

    merged = merged.reindex(columns=meta["unique_names"])
    merged.index = pd.to_datetime(merged.index, errors="coerce")
    merged = merged[merged.index.notna()].sort_index()
    merged = merged.ffill().fillna(0.0)
    return merged


def prepare_sheet_data(file_path: str, sheet_idx: int) -> Tuple[pd.DataFrame, Dict]:
    local_df, meta = load_sheet_with_meta(file_path, sheet_idx)
    mode = DATA_SOURCE_MODE.upper()

    if mode == "LOCAL_ONLY":
        return local_df, meta

    start_date, end_date = infer_fetch_window(local_df, mode)
    if not start_date or not end_date:
        return local_df, meta

    api_df, errors = fetch_ths_close_prices(meta, start_date, end_date)
    final_df = merge_local_and_api(local_df, api_df, meta, mode)

    if errors:
        print(f"⚠️ Sheet{sheet_idx + 1} 有 {len(errors)} 个标的更新失败，将沿用本地历史。")

    return final_df, meta

def calculate_residual_momentum(asset_prices: pd.DataFrame, market_prices: pd.Series, window: int) -> tuple[pd.DataFrame, pd.DataFrame]:
    """
    残差动量采用“两步法”：
    1) 用过去 window 日收益率回归，估计 alpha/beta；
    2) 用该 alpha/beta 计算当日(样本外)残差收益；
    3) 在信号日，对过去 window 日的样本外残差做 rolling(sum/std) 作为残差动量。

    这样可以避免“在同一窗口内一边回归一边累计残差”导致的两个问题：
    - 含截距 OLS 的样本内残差和天然接近 0；
    - 再把这些残差做几何连乘，容易因为波动拖累而整体偏负。
    """
    print(f"⏳ 计算特质残差动量 (基准={market_prices.name}, 窗口={window}, 口径=样本外残差 rolling(sum/std))...")
    asset_rets = asset_prices.pct_change().fillna(0.0)
    market_rets = market_prices.pct_change().fillna(0.0)

    idx = asset_rets.index.intersection(market_rets.index)
    asset_rets = asset_rets.loc[idx]
    market_rets = market_rets.loc[idx]

    resid_oos = pd.DataFrame(index=idx, columns=asset_rets.columns, dtype=float)

    x = market_rets.values.astype(float)
    Y = asset_rets.values.astype(float)
    n_assets = Y.shape[1]

    for t in range(window, len(idx)):
        y_est = Y[t-window:t, :]
        x_est = x[t-window:t]

        x_mean = np.mean(x_est)
        y_mean = np.mean(y_est, axis=0)

        numerator = np.sum((x_est[:, None] - x_mean) * (y_est - y_mean), axis=0)
        denominator = np.sum((x_est - x_mean) ** 2)

        beta = np.zeros(n_assets)
        if denominator != 0:
            beta = numerator / denominator
        alpha = y_mean - beta * x_mean

        # 用上一段估计到的 alpha/beta 去解释“当日”收益，得到样本外残差
        resid_oos.iloc[t] = Y[t, :] - (alpha + beta * x[t])

    roll_sum = resid_oos.rolling(window=window, min_periods=max(20, window // 2)).sum()
    roll_std = resid_oos.rolling(window=window, min_periods=max(20, window // 2)).std()

    scores = roll_sum.divide(roll_std.replace(0.0, np.nan))
    return scores, resid_oos


def compute_rebalance_dates(index: pd.DatetimeIndex, mode: str) -> set:
    idx = pd.DatetimeIndex(index)
    idx = idx[~pd.isna(idx)]
    if mode.upper() == "MONTHLY":
        s = pd.Series(idx, index=idx)
        month_starts = s.groupby([s.index.year, s.index.month]).min().dropna()
        return set(month_starts.dt.strftime("%Y-%m-%d"))
    raise ValueError(f"Unsupported REBALANCE_MODE: {mode}")


def apply_no_trade_band(weight_diff: pd.Series, band: float) -> pd.Series:
    if band <= 0:
        return weight_diff
    d = weight_diff.copy()
    d[d.abs() < band] = 0.0
    return d


def apply_asymmetric_l1_shrink(weight_diff: pd.Series, lam_buy: float, lam_sell: float) -> pd.Series:
    if lam_buy <= 0 and lam_sell <= 0:
        return weight_diff
    d = weight_diff.astype(float).copy()
    buy = d.clip(lower=0.0)
    sell = (-d).clip(lower=0.0)
    if lam_buy > 0:
        buy = (buy - lam_buy).clip(lower=0.0)
    if lam_sell > 0:
        sell = (sell - lam_sell).clip(lower=0.0)
    return buy - sell


def get_metrics(nav: pd.Series, rf: float) -> dict:
    if len(nav) < 10:
        return {}
    total_ret = nav.iloc[-1] - 1
    days = (nav.index[-1] - nav.index[0]).days
    ann_ret = (nav.iloc[-1] ** (365 / days)) - 1 if days > 0 else 0.0
    dr = nav.pct_change().dropna()
    ann_vol = dr.std() * np.sqrt(252)
    sharpe = (ann_ret - rf) / ann_vol if ann_vol > 0 else 0.0
    cm = nav.cummax()
    dd = (nav - cm) / cm
    return {"累计收益": total_ret, "年化收益": ann_ret, "年化波动": ann_vol, "最大回撤": dd.min(), "夏普比率": sharpe}


def get_annual_returns(nav: pd.Series) -> pd.Series:
    yearly = nav.resample("A").last()
    ar = yearly.pct_change()
    if len(ar) > 0:
        ar.iloc[0] = yearly.iloc[0] / nav.iloc[0] - 1
    ar.index = ar.index.year
    ar.index.name = "Year"
    return ar


def force_clean_small_positions(w: pd.Series, threshold: float) -> tuple[pd.Series, float]:
    """
    把卫星内部权重中小于 threshold 的零碎仓位清零，并把释放权重平均分给其余有效持仓。
    返回：(清理后的权重, 被清理掉的总权重)
    """
    w = w.fillna(0.0).astype(float).copy()
    if threshold <= 0:
        return w, 0.0

    small_mask = (w > 0) & (w < threshold)
    if not small_mask.any():
        return w, 0.0

    freed = float(w[small_mask].sum())
    w[small_mask] = 0.0

    keep_mask = w >= threshold
    keep_count = int(keep_mask.sum())
    if keep_count > 0 and freed > 0:
        w.loc[keep_mask] += freed / keep_count

    w[w < 0] = 0.0
    return w, freed


def get_effective_tiny_threshold_internal(sat_bucket_weight_total: float) -> float:
    """
    计算用于清理零碎仓的“卫星内部阈值”。
    默认保留原内部阈值；若启用总资产口径阈值，则换算到卫星内部口径后取更严格者。
    """
    thr = float(max(0.0, MIN_HOLD_THRESHOLD_INTERNAL))
    if USE_TOTAL_ASSET_THRESHOLD_FOR_CLEAN and sat_bucket_weight_total > 1e-12:
        thr_total_as_internal = float(MIN_HOLD_THRESHOLD_TOTAL) / float(sat_bucket_weight_total)
        thr = max(thr, thr_total_as_internal)
    return thr

def main():
    reset_ths_fetch_summary()
    print(f"📂 读取文件: {INPUT_FILE}")

    # 背景：THS 对不同类型指数（如 .TI 合成指数）存在数据发布延迟，导致 bench 末尾
    #       日期落后于 pool，直接 intersection 会丢失最新调仓信号。
    #       ffill 用 bench 最后一个已知收盘价填充新增交易日，保持数据连贯性。
    if bench.index.max() < pool.index.max():
        pool_only_dates = pool.index.difference(bench.index)
        print(
            f"⚠️  bench 日期落后 pool {len(pool_only_dates)} 个交易日"
            f"（{pool_only_dates.min().date()} ~ {pool_only_dates.max().date()}），"
            f"对缺失日期执行 ffill 对齐。"
        )
    bench = bench.reindex(pool.index).ffill().fillna(0.0)

    if SAVE_UPDATED_INPUT and DATA_SOURCE_MODE.upper() in {"LOCAL_PLUS_THS", "THS_ONLY"}:
        try:
            save_updated_input_workbook(UPDATED_INPUT_FILE, [(pool, pool_meta), (bench, bench_meta)])
            print(f"💾 更新后的输入数据已保存：{UPDATED_INPUT_FILE}")
        except Exception as e:
            print(f"⚠️ 保存更新后的输入文件失败：{e}")

    ths_validation_df = get_ths_fetch_summary_df()
    if PRINT_THS_VALIDATION_TABLE and DATA_SOURCE_MODE.upper() in {"LOCAL_PLUS_THS", "THS_ONLY"}:
        print_ths_fetch_summary_table(ths_validation_df)

    # bench 已与 pool 对齐，intersection 等价于 pool.index，保留原逻辑兼容性
    idx = pool.index.intersection(bench.index).sort_values()
    pool = pool.loc[idx]
    bench = bench.loc[idx]

    if MARKET_BENCH_NAME in bench.columns:
        mkt = bench[MARKET_BENCH_NAME]
    else:
        mkt = bench.iloc[:, 0]

    if HOLD_INDEX_NAME in bench.columns:
        core_ret = bench[HOLD_INDEX_NAME].pct_change().fillna(0.0)
    elif HOLD_INDEX_NAME in pool.columns:
        core_ret = pool[HOLD_INDEX_NAME].pct_change().fillna(0.0)
    else:
        core_ret = pd.Series(0.0, index=pool.index)

    resid_mom, resid_oos = calculate_residual_momentum(pool, mkt, RESIDUAL_WINDOW)
    ma_fast = pool.rolling(MA_FAST).mean()
    ma_slow = pool.rolling(MA_SLOW).mean()

    daily_rets = pool.pct_change().fillna(0.0)
    rebalance_days = compute_rebalance_dates(pool.index, REBALANCE_MODE)

    # 卫星内部限重：按“投资资产比例”换算（更符合现金约束下的真实约束）
    investable_center = 1.0 - CASH_RESERVE_RATIO
    sat_center = investable_center * (1.0 - CORE_TARGET_RATIO) if CORE_RATIO_ON_INVESTABLE else (1.0 - CASH_RESERVE_RATIO - CORE_TARGET_RATIO)
    sat_center = max(sat_center, 1e-12)
    max_w_internal = MAX_SINGLE_WEIGHT / sat_center

    n = len(pool)
    core_val = np.zeros(n, dtype=float)
    sat_val  = np.zeros(n, dtype=float)
    cash_val = np.zeros(n, dtype=float)
    total_val = np.zeros(n, dtype=float)

    # 初始分配：现金1%，剩余按核心/卫星目标分
    cash_val[0] = INITIAL_CAPITAL * CASH_RESERVE_RATIO
    investable0 = INITIAL_CAPITAL - cash_val[0]
    if CORE_RATIO_ON_INVESTABLE:
        core_val[0] = investable0 * CORE_TARGET_RATIO
        sat_val[0]  = investable0 - core_val[0]
    else:
        core_val[0] = INITIAL_CAPITAL * CORE_TARGET_RATIO
        sat_val[0]  = INITIAL_CAPITAL - core_val[0] - cash_val[0]
    total_val[0] = core_val[0] + sat_val[0] + cash_val[0]

    # 卫星权重（相对 sat_val）
    w = pd.Series(0.0, index=pool.columns)
    pos_hist = pd.DataFrame(0.0, index=pool.index, columns=pool.columns)

    # 滞后执行
    lag_timer = 0
    pending = None  # {"target_w": Series, "risk_off": bool}

    # 滞回状态
    entry_streak = {a: 0 for a in pool.columns}
    exit_streak  = {a: 0 for a in pool.columns}

    # 换手与年度预算
    trades = []
    is_first_trade = True

    curr_year = pool.index[0].year
    sum_assets_year = 0.0
    count_days_year = 0
    cum_trade_val_year = 0.0

    # 【新增】用于保存最后一次调仓信号的变量
    last_signal_info = None
    last_cleanup_info = None

    print(
        f"🚀 回测开始 | CORE_MODE={CORE_MODE} CORE_TARGET={CORE_TARGET_RATIO:.0%} "
        f"(on_investable={CORE_RATIO_ON_INVESTABLE}) | CASH={CASH_RESERVE_RATIO:.0%} | "
        f"T+{TRADING_LAG} | 年化换手上限={TURNOVER_CAP_ANNUAL:.0%} | "
        f"L1(buy/sell)={L1_LAMBDA_BUY}/{L1_LAMBDA_SELL}"
    )

    for i in range(1, n):
        date = pool.index[i]
        date_str = date.strftime("%Y-%m-%d")

        # 年度重置
        if date.year != curr_year:
            curr_year = date.year
            sum_assets_year = 0.0
            count_days_year = 0
            cum_trade_val_year = 0.0

        # 1) 更新净值（按昨日权重）
        core_val[i] = core_val[i-1] * (1 + core_ret.iloc[i])
        sat_r = float((w * daily_rets.iloc[i]).sum())
        sat_val[i]  = sat_val[i-1] * (1 + sat_r)
        cash_val[i] = cash_val[i-1] * (1 + CASH_DAILY_RET)
        total_val[i] = core_val[i] + sat_val[i] + cash_val[i]

        # 年内平均资产（迄今）
        sum_assets_year += total_val[i]
        count_days_year += 1
        avg_asset = sum_assets_year / count_days_year

        # 2) 执行 pending（到期）
        if lag_timer > 0:
            lag_timer -= 1
            if lag_timer == 0 and pending is not None:
                pre_core = core_val[i]
                pre_sat  = sat_val[i]
                pre_cash = cash_val[i]
                tot = total_val[i]
                if tot <= 0:
                    pending = None
                else:
                    # === 2.1 顶层强制：现金 + 核心（与系统一致，不能缩放） ===
                    cash_target = tot * CASH_RESERVE_RATIO
                    investable = tot - cash_target

                    if CORE_MODE.upper() == "FORCED_50":
                        core_ratio_target = CORE_TARGET_RATIO
                    elif CORE_MODE.upper() == "BAND":
                        # band 模式：在投资资产口径下做 band
                        core_ratio_now = (pre_core / max(investable, 1e-12)) if CORE_RATIO_ON_INVESTABLE else (pre_core / tot)
                        core_ratio_target = core_ratio_now
                        if core_ratio_now < CORE_BAND_LOW:
                            core_ratio_target = CORE_BAND_LOW
                        elif core_ratio_now > CORE_BAND_HIGH:
                            core_ratio_target = CORE_BAND_HIGH
                    else:
                        raise ValueError("CORE_MODE 仅支持 FORCED_50 或 BAND")

                    if CORE_RATIO_ON_INVESTABLE:
                        core_target = investable * core_ratio_target
                        sat_target  = investable - core_target
                    else:
                        core_target = tot * core_ratio_target
                        sat_target  = tot - core_target - cash_target

                    # 顶层 diffs（满足三桶和为0）
                    d_cash = cash_target - pre_cash
                    d_core = core_target - pre_core
                    d_sat  = sat_target  - pre_sat

                    # 数值漂移纠偏：让三者严格求和为0
                    drift = d_cash + d_core + d_sat
                    if abs(drift) > 1e-8:
                        d_sat -= drift

                    # 资金约束：不允许出现负值（这里做轻微裁剪，极端情况下可能导致比率不精确）
                    new_cash = pre_cash + d_cash
                    new_core = pre_core + d_core
                    new_sat  = pre_sat  + d_sat
                    if new_cash < 0:
                        # 缺口从卫星补
                        deficit = -new_cash
                        new_cash = 0.0
                        new_sat -= deficit
                    if new_core < 0:
                        deficit = -new_core
                        new_core = 0.0
                        new_sat -= deficit
                    if new_sat < 0:
                        deficit = -new_sat
                        new_sat = 0.0
                        # 反向从现金补（尽量不破现金约束，但极端情况只能允许）
                        new_cash -= deficit
                        new_cash = max(new_cash, 0.0)

                    # 顶层交易的 buy+sell 名义额 = |Δcash|+|Δcore|+|Δsat|（L1范数）
                    top_trade_val = abs(new_cash - pre_cash) + abs(new_core - pre_core) + abs(new_sat - pre_sat)

                    cash_val[i] = new_cash
                    core_val[i] = new_core
                    sat_val[i]  = new_sat
                    total_val[i] = cash_val[i] + core_val[i] + sat_val[i]

                    if top_trade_val > 0:
                        trades.append({"date": date, "amt": top_trade_val, "is_initial": is_first_trade, "type": "top_rebalance"})
                        is_first_trade = False
                        cum_trade_val_year += top_trade_val

                    # === 2.2 卫星内部交易：受预算约束 + L1收缩 ===
                    effective_cap = TURNOVER_CAP_ANNUAL * TURNOVER_SAFETY_BUFFER
                    remaining = 2.0 * effective_cap * avg_asset - cum_trade_val_year
                    remaining = max(0.0, remaining)

                    target_w = pending["target_w"].reindex(pool.columns).fillna(0.0)
                    if pending.get("risk_off", False) and RISK_OFF_FORCE_LIQUIDATE:
                        d_w_full = -w.copy()
                    else:
                        d_w_full = target_w - w
                        d_w_full = apply_asymmetric_l1_shrink(d_w_full, L1_LAMBDA_BUY, L1_LAMBDA_SELL)
                        d_w_full = apply_no_trade_band(d_w_full, WEIGHT_NO_TRADE_BAND)
                        d_w_full = PARTIAL_KAPPA * d_w_full

                    sat_trade_val_exec = 0.0
                    if not (d_w_full.abs().sum() <= MIN_TRADE_THRESHOLD or remaining <= 0 or sat_val[i] <= 0):
                        sat_after = sat_val[i]
                        sat_trade_w = float(d_w_full.clip(lower=0.0).sum() + (-d_w_full).clip(lower=0.0).sum())
                        sat_trade_val_full = sat_trade_w * sat_after

                        s = min(1.0, remaining / sat_trade_val_full) if sat_trade_val_full > 0 else 0.0
                        d_w = s * d_w_full

                        sat_trade_w_exec = float(d_w.clip(lower=0.0).sum() + (-d_w).clip(lower=0.0).sum())
                        sat_trade_val_exec = sat_trade_w_exec * sat_after

                        if sat_trade_val_exec > 1e-12:
                            w = w.add(d_w, fill_value=0.0)
                            w[w < 0] = 0.0
                            trades.append({"date": date, "amt": sat_trade_val_exec, "is_initial": False, "type": "satellite_trade"})
                            cum_trade_val_year += sat_trade_val_exec

                    remaining_after_sat_trade = max(0.0, remaining - sat_trade_val_exec)
                    if FORCE_DROP_TINY_POSITIONS and sat_val[i] > 0:
                        sat_bucket_weight_total = sat_val[i] / max(total_val[i], 1e-12)
                        tiny_threshold_internal = get_effective_tiny_threshold_internal(sat_bucket_weight_total)
                        w_candidate, freed_weight = force_clean_small_positions(w, tiny_threshold_internal)
                        clean_trade_amt = 2.0 * freed_weight * sat_val[i] if freed_weight > 1e-12 else 0.0

                        allow_clean = False
                        if clean_trade_amt > 1e-12:
                            if CLEAN_TINY_POSITIONS_ONLY_IF_BUDGET_OK:
                                allow_clean = clean_trade_amt <= remaining_after_sat_trade + 1e-12
                            else:
                                allow_clean = True

                        last_cleanup_info = {
                            "date": date,
                            "date_str": date_str,
                            "sat_bucket_weight_total": float(sat_bucket_weight_total),
                            "tiny_threshold_internal": float(tiny_threshold_internal),
                            "tiny_threshold_total": float(tiny_threshold_internal * sat_bucket_weight_total),
                            "freed_weight_internal": float(freed_weight),
                            "clean_trade_amt": float(clean_trade_amt),
                            "remaining_after_sat_trade": float(remaining_after_sat_trade),
                            "executed": bool(allow_clean),
                        }

                        if allow_clean:
                            w = w_candidate
                            trades.append({
                                "date": date,
                                "amt": clean_trade_amt,
                                "is_initial": False,
                                "type": "satellite_small_pos_clean"
                            })
                            cum_trade_val_year += clean_trade_amt

                        elif clean_trade_amt > 1e-12:
                            print(f"⚠️  {date_str} 零碎仓清理跳过：需额外换手 {clean_trade_amt:.2f}，剩余预算仅 {remaining_after_sat_trade:.2f}")

                    pending = None

        # 记录持仓
        pos_hist.iloc[i] = w.values

        # 3) 生成新 pending（调仓日且无 pending 且无 lag）
        if (date_str not in rebalance_days) or (lag_timer != 0) or (pending is not None):
            continue

        # ===== 月度信号逻辑 =====
        # 调入：残差动量排名前三，且位于 20/60 日均线上方
        # 调出：当前持仓跌破均线，或排名跌出“前三 + BUFFER_RANK”的缓冲区
        price = pool.loc[date]
        current = w[w > 1e-8].index.tolist()

        scores = resid_mom.loc[date].dropna().copy()
        if scores.empty:
            risk_off = True
            final_targets = []
        else:
            ranks = scores.sort_values(ascending=False).index.tolist()
            rank_map = {a: i + 1 for i, a in enumerate(ranks)}

            def is_above_ma(asset: str) -> bool:
                return (
                    pd.notna(ma_fast.loc[date, asset]) and
                    pd.notna(ma_slow.loc[date, asset]) and
                    pd.notna(price[asset]) and
                    (price[asset] > ma_fast.loc[date, asset]) and
                    (price[asset] > ma_slow.loc[date, asset])
                )

            entry_candidates = [
                a for a in ranks
                if rank_map.get(a, 9999) <= ENTRY_RANK and is_above_ma(a)
            ]

            kept = []
            for a in current:
                within_buffer = rank_map.get(a, 9999) <= EXIT_RANK
                if within_buffer and is_above_ma(a):
                    kept.append(a)
